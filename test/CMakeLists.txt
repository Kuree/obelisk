function(talos_add_generated_process_object output_var directory input_file)
  set(generated_dir "${CMAKE_CURRENT_BINARY_DIR}/${directory}")
  set(generated_mlir "${generated_dir}/process.mlir")
  set(generated_raw_llvm "${generated_dir}/process.raw.ll")
  set(generated_llvm "${generated_dir}/process.ll")
  set(generated_object "${generated_dir}/process.o")
  set(generated_opt_options)
  set(generated_bytecode_options "vpi=off")
  cmake_parse_arguments(ARG "REQUIRE_BYTECODE" "VPI" "" ${ARGN})
  if(ARG_UNPARSED_ARGUMENTS)
    message(FATAL_ERROR
      "unknown generated bytecode options: ${ARG_UNPARSED_ARGUMENTS}")
  endif()
  if(ARG_VPI)
    if(NOT ARG_VPI MATCHES "^(off|read|write)$")
      message(FATAL_ERROR "unknown generated VPI profile: ${ARG_VPI}")
    endif()
    set(generated_bytecode_options "vpi=${ARG_VPI}")
  endif()
  if(ARG_REQUIRE_BYTECODE)
    string(APPEND generated_bytecode_options " require-bytecode")
  endif()
  if(LLVM_USE_SANITIZER MATCHES "(^|;)Thread($|;)")
    # The official prebuilt LLVM/MLIR SDK is not TSAN-instrumented. Keep its
    # internal verifier thread pool from crossing the instrumentation boundary
    # while generating runtime test fixtures; the runtime tests themselves
    # remain multithreaded.
    list(APPEND generated_opt_options --mlir-disable-threading)
  endif()
  file(MAKE_DIRECTORY "${generated_dir}")
  add_custom_command(
    OUTPUT "${generated_object}"
    COMMAND $<TARGET_FILE:talos-opt>
            ${generated_opt_options}
            "${CMAKE_CURRENT_SOURCE_DIR}/${input_file}"
            "--pass-pipeline=builtin.module(simulation.design(simulation.func(talos-sim-thread-suspension)),encode-talos-sim-to-bytecode{${generated_bytecode_options}},convert-talos-sim-processes-to-llvm-coroutines)"
            -o "${generated_mlir}"
    COMMAND $<TARGET_FILE:talos-translate> --mlir-to-llvmir
            "${generated_mlir}" -o "${generated_raw_llvm}"
    COMMAND "${TALOS_LLVM_DIST_DIR}/bin/opt"
            "-passes=default<O3>"
            -S "${generated_raw_llvm}" -o "${generated_llvm}"
    COMMAND "${TALOS_LLVM_DIST_DIR}/bin/llc"
            "-mtriple=${TALOS_TARGET_TRIPLE}" -filetype=obj
            -relocation-model=pic "${generated_llvm}"
            -o "${generated_object}"
    DEPENDS
    talos-opt
    talos-translate
    "${CMAKE_CURRENT_SOURCE_DIR}/${input_file}"
    VERBATIM
  )
  set_source_files_properties("${generated_object}" PROPERTIES
    EXTERNAL_OBJECT TRUE
    GENERATED TRUE
  )
  set(${output_var} "${generated_object}" PARENT_SCOPE)
endfunction()
